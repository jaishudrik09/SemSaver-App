    // --- "WHAT NEEDS YOUR ATTENTION?" DETERMINISTIC LOGIC ---
    function renderAttentionEngine() {
      const container = document.getElementById('attention-items-list');
      if (!container) return;

      const alerts = [];

      // 1. Check low attendance (< 75%)
      state.subjects.forEach(s => {
        const pct = s.total > 0 ? ((s.attended / s.total) * 100).toFixed(0) : 0;
        if (pct < 75) {
          const needed = Math.ceil((0.75 * s.total - s.attended) / 0.25);
          alerts.push({
            type: 'attendance',
            icon: 'alert-triangle',
            iconColor: 'text-rose-400',
            borderColor: 'border-rose-800/40 bg-rose-950/20',
            title: `Attendance warning: ${s.name} (${pct}%)`,
            desc: `Below 75% threshold. Attend next ${needed} class${needed > 1 ? 'es' : ''} to recover.`
          });
        }
      });

      // 2. Check urgent tasks (due today or within 48 hrs)
      const now = new Date();
      state.tasks.filter(t => !t.completed).forEach(t => {
        const due = new Date(t.dueDate);
        const diffHrs = (due - now) / (1000 * 60 * 60);
        if (diffHrs <= 24 && diffHrs > 0) {
          alerts.push({
            type: 'task',
            icon: 'clock',
            iconColor: 'text-amber-400',
            borderColor: 'border-amber-800/40 bg-amber-950/20',
            title: `Deadline in < 24 hrs: ${t.title}`,
            desc: `Due at ${due.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })} (${t.subject})`
          });
        }
      });

      // 3. Check exams coming in <= 14 days
      state.exams.forEach(e => {
        const examDate = new Date(e.date);
        const diffDays = Math.ceil((examDate - now) / (1000 * 60 * 60 * 24));
        if (diffDays <= 14 && diffDays >= 0) {
          alerts.push({
            type: 'exam',
            icon: 'flame',
            iconColor: 'text-purple-400',
            borderColor: 'border-purple-800/40 bg-purple-950/20',
            title: `Exam in ${diffDays} days: ${e.subject}`,
            desc: `${e.name} in room ${e.room}`
          });
        }
      });

      if (alerts.length === 0) {
        container.innerHTML = `
          <div class="col-span-full py-2 px-3 rounded-xl bg-emerald-950/20 border border-emerald-800/30 text-emerald-300 text-xs flex items-center space-x-2">
            <i data-lucide="check-circle-2" class="w-4 h-4 text-emerald-400"></i>
            <span>No critical warnings right now! Attendance is healthy and no deadlines imminent.</span>
          </div>
        `;
        return;
      }

      container.innerHTML = alerts.slice(0, 3).map(a => `
        <div class="p-2.5 sm:p-3 rounded-xl border ${a.borderColor} flex items-center space-x-2.5 transition">
          <i data-lucide="${a.icon}" class="w-4 h-4 shrink-0 ${a.iconColor}"></i>
          <div class="min-w-0 flex-1">
            <h4 class="text-xs font-semibold text-white leading-tight truncate">${a.title}</h4>
            <p class="text-[11px] text-slate-400 truncate mt-0.5">${a.desc}</p>
          </div>
        </div>
      `).join('');
      lucide.createIcons();
    }

    // --- RENDER TASKS & PRACTICALS ---
    function renderTasksList() {
      const container = document.getElementById('tasks-container');
      const filter = document.getElementById('tasks-filter').value;
      const now = new Date();

      let items = state.tasks;
      if (filter === 'pending') items = items.filter(t => !t.completed);
      if (filter === 'completed') items = items.filter(t => t.completed);
      if (filter === 'assignment') items = items.filter(t => t.type === 'assignment');
      if (filter === 'practical') items = items.filter(t => t.type === 'practical');

      if (items.length === 0) {
        container.innerHTML = `
          <div class="text-center py-10 rounded-2xl bg-surface-900 border border-slate-800 text-slate-400 text-xs">
            <i data-lucide="inbox" class="w-8 h-8 mx-auto mb-2 text-slate-600"></i>
            <p>No tasks matching this filter.</p>
          </div>
        `;
        return;
      }

      container.innerHTML = items.map(t => {
        const due = new Date(t.dueDate);
        const diffHrs = (due - now) / (1000 * 60 * 60);
        let dueLabel = "";

        if (t.completed) {
          dueLabel = '<span class="text-[10px] px-2 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 font-semibold">Completed</span>';
        } else if (diffHrs < 0) {
          dueLabel = '<span class="text-[10px] px-2 py-0.5 rounded-full bg-rose-500 text-white font-semibold">Overdue</span>';
        } else if (diffHrs <= 24) {
          dueLabel = '<span class="text-[10px] px-2 py-0.5 rounded-full bg-rose-500/15 text-rose-400 border border-rose-500/20 font-bold">Due today</span>';
        } else if (diffHrs <= 48) {
          dueLabel = '<span class="text-[10px] px-2 py-0.5 rounded-full bg-amber-500/15 text-amber-400 border border-amber-500/20 font-semibold">Due tomorrow</span>';
        } else {
          dueLabel = `<span class="text-[10px] px-2 py-0.5 rounded-full bg-slate-800 text-slate-300 font-medium">Due in ${Math.round(diffHrs / 24)} days</span>`;
        }

        return `
          <div class="p-4 rounded-xl border border-slate-800 bg-surface-900 flex items-center justify-between gap-3 ${t.completed ? 'opacity-60' : ''}">
            <div class="flex items-center space-x-3.5 min-w-0">
              <button onclick="toggleTask('${t.id}')" class="w-5 h-5 rounded-md border ${t.completed ? 'bg-indigo-600 border-indigo-600 text-white' : 'border-slate-600 hover:border-indigo-400'} flex items-center justify-center shrink-0 transition">
                ${t.completed ? '<i data-lucide="check" class="w-3.5 h-3.5"></i>' : ''}
              </button>
              <div class="min-w-0">
                <div class="flex items-center space-x-2 flex-wrap gap-y-1">
                  <h4 class="text-xs sm:text-sm font-bold text-white ${t.completed ? 'line-through text-slate-400' : ''}">${t.title}</h4>
                  <span class="text-[10px] px-2 py-0.2 rounded bg-slate-800 text-slate-300 font-mono">${t.subject}</span>
                  <span class="text-[10px] px-1.5 py-0.2 rounded bg-indigo-500/10 text-indigo-300 capitalize">${t.type}</span>
                </div>
                <p class="text-[11px] text-slate-500 mt-1 flex items-center space-x-1.5">
                  <i data-lucide="calendar" class="w-3 h-3"></i>
                  <span>${due.toLocaleDateString([], { month: 'short', day: 'numeric' })} at ${due.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}</span>
                </p>
              </div>
            </div>

            <div class="flex items-center space-x-2.5 shrink-0">
              ${dueLabel}
              <button onclick="deleteTask('${t.id}')" title="Delete task" class="text-slate-500 hover:text-rose-400 p-1 transition">
                <i data-lucide="trash-2" class="w-4 h-4"></i>
              </button>
            </div>
          </div>
        `;
      }).join('');
      lucide.createIcons();
    }

    function adjustSimBunk(delta) {
      simBunkDelta = Math.max(1, Math.min(10, simBunkDelta + delta));
      const el = document.getElementById('sim-bunk-count');
      if (el) el.innerText = simBunkDelta;

    function runBunkForecast() {
      const select = document.getElementById('sim-subject-select');
      const outcomeText = document.getElementById('sim-outcome-text');
      if (!select || !outcomeText || !state.subjects.length) return;

      const sub = state.subjects.find(s => s.id === select.value) || state.subjects[0];
      if (!sub) return;

      const futureAttended = sub.attended;
      const futureTotal = sub.total + simBunkDelta;
      const futurePct = futureTotal > 0 ? ((futureAttended / futureTotal) * 100).toFixed(1) : 0;

      if (futurePct >= 75) {
        outcomeText.innerHTML = `<span class="text-emerald-400 font-mono font-bold">${futurePct}% (SAFE)</span> · You can safely skip ${simBunkDelta} class${simBunkDelta > 1 ? 'es' : ''}!`;
      } else {
        const needed = Math.ceil((0.75 * futureTotal - futureAttended) / 0.25);
        outcomeText.innerHTML = `<span class="text-rose-400 font-mono font-bold">${futurePct}% (DEBAR RISK!)</span> · You would drop below 75% target.`;
      }
    }

      const container = document.getElementById('exams-container');

      container.innerHTML = state.exams.map(e => {

        const diffDays = Math.max(0, Math.ceil(diffMs / (1000 * 60 * 60 * 24)));

        return `
          <div class="p-5 rounded-2xl border border-slate-800 bg-surface-900 flex flex-col justify-between space-y-4">
            <div>
              <div class="flex items-start justify-between">

                  <p class="text-xs text-slate-400 mt-0.5">Room ${e.room} · ${examDate.toLocaleDateString([], { weekday: 'short', month: 'short', day: 'numeric' })}</p>
                </div>
              </div>

              <!-- Days countdown badge -->
              <div class="my-3 p-3 rounded-xl bg-slate-800/60 border border-slate-700/60 flex items-center justify-between">

          if (!raw) return alert("Please paste a WhatsApp message or click a demo notice above!");

          else if (lower.includes("network") || lower.includes("cn")) sub = "Computer Networks";
